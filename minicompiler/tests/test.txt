// 娴嬭瘯閫昏緫杩愮畻绗?
int main() {
    int a = 10;
    int b = 20;
    int c = 0;
    int result;
    
    // 娴嬭瘯閫昏緫涓?&&)
    if (a > 5 && b < 30) {
        result = 1;
    }
    
    // 娴嬭瘯閫昏緫鎴?||)
    if (a < 5 || b > 15) {
        result = 2;
    }
    
    // 娴嬭瘯閫昏緫闈?!)
    if (!c) {  // c 涓?0 鏃舵潯浠朵负鐪?
        result = 3;
    }
    
    // 娴嬭瘯澶嶆潅閫昏緫琛ㄨ揪寮?
    if ((a > 5 && b < 30) || (!c && a != b)) {
        result = 4;
    }
    
    // 娴嬭瘯鐭矾閫昏緫
    if (a < 0 && b/0 > 0) {  // b/0 涓嶄細鎵ц锛屽洜涓?a < 0 宸茬粡涓哄亣
        result = 5;
    }
    
    // 娴嬭瘯浼樺厛绾?
    if (!a > 0 || b > 15 && c == 0) {
        result = 6;
    }
    
    return result;
}
