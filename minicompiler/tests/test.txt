// 测试逻辑运算符
int main() {
    int a = 10;
    int b = 20;
    int c = 0;
    int result;
    
    // 测试逻辑与(&&)
    if (a > 5 && b < 30) {
        result = 1;
    }
    
    // 测试逻辑或(||)
    if (a < 5 || b > 15) {
        result = 2;
    }
    
    // 测试逻辑非(!)
    if (!c) {  // c 为 0 时条件为真
        result = 3;
    }
    
    // 测试复杂逻辑表达式
    if ((a > 5 && b < 30) || (!c && a != b)) {
        result = 4;
    }
    
    // 测试短路逻辑
    if (a < 0 && b/0 > 0) {  // b/0 不会执行，因为 a < 0 已经为假
        result = 5;
    }
    
    // 测试优先级
    if (!a > 0 || b > 15 && c == 0) {
        result = 6;
    }
    
    return result;
}
